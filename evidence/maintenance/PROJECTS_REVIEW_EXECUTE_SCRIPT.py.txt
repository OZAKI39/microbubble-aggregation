def statset(root,exclude=None):

def tree_bytes(p):

def writable_selected(paths):

def process_hits(roots):

  ls=[proc/'cwd',proc/'exe']
  try:ls+=list((proc/'fd').iterdir())
  except OSError:pass
  for x in ls:
   try:t=os.readlink(x)
   except OSError:continue
   if any(t==r or t.startswith(r+'/') for r in roots):hits.append({'pid':proc.name,'reference':str(x),'target':t})

   for line in (proc/'maps').read_text().splitlines():

    if any(t==r or t.startswith(r+'/') for r in roots):hits.append({'pid':proc.name,'reference':'maps','target':t})
  except OSError:pass
 return hits
