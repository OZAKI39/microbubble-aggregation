"""Can a standing wave tear a 3D cluster apart? (2026-11-07, small test, measurement only)

usage (repository root):  python NODE_RUNS.py OUT_DIR
Starts: random_start seeds 23, 14, 21, 16, 17, 7 (20, 32, 50, 76, 99, 122 bubbles, the starts of
evidence/soft3d_20261107/). Drive: the opposite pair X- and X+ at 2.5 MHz for 1.5 s, the stronger
channel at the source cap (6000 Pa-eq) and the other scaled so that both give the same pressure
amplitude at the centroid of the start, with the relative phase that puts at that centroid
  node:     a pressure node (the two waves cancel there), or
  antinode: a pressure antinode (the two waves add up there).
Adhesion: default (x1) for both; also none (x0) for the node.
Everything else is the current default. The project code is not changed: only the run's drive is
replaced in this process (microbubble.control.random_actions), and every force evaluation is
summarised by a wrapper (as in evidence/soft3d_20261107/SOFT3D_RUNS.py.txt).
Writes OUT_DIR/<name>/run, OUT_DIR/<name>/compact.npz (camera frames at 60/s, positions every
10 ms, per-step contact counts and force summaries), OUT_DIR/PROFILES.npz (pressure amplitude along
x through each centroid for both settings) and OUT_DIR/RUNS.json.
"""

import json
import sys
from pathlib import Path

import h5py
import numpy as np

import microbubble.control as control
import microbubble.forces as forces
from microbubble.acoustics import direct_quadrature_order
from microbubble.initial_conditions import random_start
from workflows.f10ag import load_pretrain_policy, pretrain_context, run

STARTS = [23, 14, 21, 16, 17, 7]
CASES = [("node", "x1", 1.0), ("node", "x0", 0.0), ("antinode", "x1", 1.0)]
FREQUENCY_HZ = 2.5e6
DURATION_S = 1.5
CAMERA_FPS = 60.0
X_MINUS, X_PLUS = 2, 3  # channel order Y-, Y+, X-, X+

out = Path(sys.argv[1])
out.mkdir(parents=True)
root = Path(".")
policy = load_pretrain_policy(root)
cap = policy["max_source_amplitude_Pa_equivalent"]
coefficient = policy["adhesion"]["coefficient_N_per_Pa2"]
context = pretrain_context(root, FREQUENCY_HZ)
order = direct_quadrature_order(FREQUENCY_HZ)
s = policy["random_state"]

drive = {}
control.random_actions = lambda duration, *rest: [dict(drive, duration=duration)]

summary = []  # one row per force evaluation of the current run
project_call = forces.CPUForces.__call__


def summarised(self, x):
    f = project_call(self, x)
    row = []
    for key in ("total_force_N", "secondary_force_N", "primary_force_N"):
        v = f[key]
        mean = v.mean(0)
        spread = np.linalg.norm(v - mean, axis=1)
        row += [np.linalg.norm(mean), np.median(spread), np.percentile(spread, 90)]
    summary.append(row)
    return f


def x_drive(h, setting):
    """Amplitudes and phases (4 channels) of X- and X+ giving equal pressure at the centroid,
    opposite (node) or equal (antinode) in phase there."""
    m, p = abs(h[X_MINUS]), abs(h[X_PLUS])
    amplitudes = np.zeros(4)
    amplitudes[X_MINUS] = cap if m <= p else cap * p / m
    amplitudes[X_PLUS] = cap if p <= m else cap * m / p
    phases = np.zeros(4)
    turn = np.pi if setting == "node" else 0.0
    phases[X_PLUS] = np.mod(np.angle(h[X_MINUS]) - np.angle(h[X_PLUS]) + turn, 2 * np.pi)
    return amplitudes, phases


forces.CPUForces.__call__ = summarised
rows, profiles = {}, {}
for seed in STARTS:
    x0, _, start = random_start(seed, context.cavity.lumen, context.physics.radius_m, s["wall_margin_m"],
                                s["bubble_count_range"], s["aspect_range"])
    centroid = x0.mean(0)
    h, _ = context.discs.evaluate(centroid[None, :], order)
    h = h[:, 0]
    line = centroid + np.outer(np.linspace(-400e-6, 400e-6, 801), [1.0, 0.0, 0.0])
    hl, _ = context.discs.evaluate(line, order)
    profiles[f"seed{seed}_x_m"] = line[:, 0] - centroid[0]
    for setting in ("node", "antinode"):
        a, p = x_drive(h, setting)
        profiles[f"seed{seed}_{setting}_abs_pressure_Pa"] = np.abs((a * np.exp(1j * p)) @ hl)
    for setting, level, factor in CASES:
        name = f"seed{seed}_{setting}_adhesion_{level}"
        folder = out / name
        folder.mkdir()
        amplitudes, phases = x_drive(h, setting)
        drive.clear()
        drive.update(amplitudes=amplitudes.tolist(), phases=phases.tolist(), frequency=FREQUENCY_HZ)
        u = amplitudes * np.exp(1j * phases)
        summary.clear()
        cfg = folder / "cfg.json"
        cfg.write_text(json.dumps(dict(fidelity="PRETRAIN", coupling="collective", duration_s=DURATION_S,
                                       save_interval_s=0.001, state_seed=seed, action_seed=0,
                                       adhesion_coefficient_N_per_Pa2=coefficient * factor)))
        r = run(root, cfg, folder / "run")
        sm = json.loads((folder / "run" / "SUMMARY.json").read_text())
        records = json.loads((folder / "run" / "STEP_RECORDS.json").read_text())
        with h5py.File(folder / "run" / "trajectory.h5") as f5:
            times, positions = f5["time_s"][()], f5["positions_m"][()]
        if not np.allclose(positions[0], x0):
            raise RuntimeError("Run start differs from the start used for the drive")
        camera_t = np.arange(0.0, times[-1] + 1e-9, 1 / CAMERA_FPS)
        k = np.minimum(np.searchsorted(times, camera_t, side="right") - 1, len(times) - 2)
        w = ((camera_t - times[k]) / (times[k + 1] - times[k]))[:, None, None]
        camera = (1 - w) * positions[k] + w * positions[k + 1]
        np.savez_compressed(
            folder / "compact.npz", camera_t=camera_t, camera_positions_m=camera,
            t10=times[::10], positions10_m=positions[::10],
            step_t=np.array([row["time_s"] for row in records]),
            active_count=np.array([row["active_count"] for row in records]),
            candidate_count=np.array([row["candidate_count"] for row in records]),
            force_summary=np.array(summary[:len(records)]),  # one force evaluation per step
            force_summary_columns=np.array(["total_mean", "total_spread_median", "total_spread_p90",
                                            "F2_mean", "F2_spread_median", "F2_spread_p90",
                                            "F1_mean", "F1_spread_median", "F1_spread_p90"]))
        rows[name] = dict(seed=seed, setting=setting, adhesion=level, bubbles=sm["N"], status=sm["status"],
                          reason=sm["reason"], completed_s=sm["completed_s"],
                          adhesion_force_N=sm["task"]["adhesion_force_N"], wall_s=sm["wall_s"],
                          amplitudes_Pa_eq=amplitudes.tolist(), phases_rad=phases.tolist(),
                          centroid_m=centroid.tolist(),
                          pressure_at_centroid_Pa=float(abs(u @ h)),
                          pressure_one_channel_at_centroid_Pa=float(abs(u[X_MINUS] * h[X_MINUS])))
        print(name, sm["N"], sm["status"], sm["reason"], sm["completed_s"], f"{sm['wall_s']:.1f} s",
              f"|P| centroid {abs(u @ h):.1f} Pa", flush=True)
np.savez_compressed(out / "PROFILES.npz", **profiles)
(out / "RUNS.json").write_text(json.dumps(rows, indent=1) + "\n")
