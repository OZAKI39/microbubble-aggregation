"""Linear response chi(R) = V1/P of a lipid-coated bubble (linearized Marmottant, as fitted by
Tu et al. 2009 JASA 126:2954 eq. 6; initial surface tension sigma0 as in the APECSS case).
Shell parameters vs radius: least-squares fits to points read off Tu et al. 2009 Fig. 4
(digitized by eye from the rendered figure; about +-10% reading error). The linear elasticity fit
crosses zero near 1.0 um; below that it is held at 0.03 N/m, Tu's smallest values (0.8-1.0 um)."""

import numpy as np

P0, RHO, C, MU, GAMMA, SIGMA0 = 101300.0, 1000.0, 1500.0, 0.002, 1.07, 0.036
# Tu 2009 Fig. 4(a): R0 (um), chi (N/m); Fig. 4(b): R0 (um), kappa_s (kg/s), filled circles (their data)
FIG4A = np.array([(0.8, .03), (1.0, .04), (1.3, .11), (1.4, .09), (1.5, .17), (1.6, .22), (1.7, .24), (1.75, .26),
                  (1.8, .30), (1.85, .27), (1.9, .35), (2.0, .43), (2.1, .50), (2.25, .50), (2.4, .65), (2.5, .70),
                  (2.7, .80), (3.0, .90), (3.25, .87)])
FIG4B = np.array([(0.8, 1e-9), (1.0, 1.5e-9), (1.3, 3e-9), (1.4, 2e-9), (1.5, 3.5e-9), (1.6, 3.2e-9), (1.7, 4e-9),
                  (1.75, 3.5e-9), (1.8, 5.5e-9), (1.85, 6.5e-9), (1.9, 6.5e-9), (2.0, 7e-9), (2.05, 8e-9),
                  (2.25, 1.6e-8), (2.5, 1.6e-8), (2.7, 2e-8), (3.0, 4e-8), (3.25, 3e-8)])
CHI_FIT = np.polyfit(FIG4A[:, 0], FIG4A[:, 1], 1)
KAPPA_FIT = np.polyfit(FIG4B[:, 0], np.log10(FIG4B[:, 1]), 1)


def tu_shell(radius_m):
    r_um = np.asarray(radius_m) * 1e6
    return np.maximum(np.polyval(CHI_FIT, r_um), 0.03), 10 ** np.polyval(KAPPA_FIT, r_um)


def chi(radius_m, frequency_hz, elasticity_n_m, shell_viscosity_kg_s):
    R0 = np.asarray(radius_m)
    w = 2 * np.pi * frequency_hz
    pg0 = P0 + 2 * SIGMA0 / R0
    stiffness = 3 * GAMMA * pg0 - 2 * SIGMA0 / R0 + 4 * elasticity_n_m / R0
    damping = 4 * MU + 4 * shell_viscosity_kg_s / R0 + 3 * GAMMA * pg0 * R0 / C
    return -4 * np.pi * R0 ** 3 / (stiffness - RHO * R0 * R0 * w * w + 1j * w * damping)
