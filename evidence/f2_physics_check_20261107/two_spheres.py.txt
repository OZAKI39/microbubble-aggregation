"""Exact incompressible potential flow around two spheres that pulsate and translate along their
line of centres (axisymmetric multipole expansion with re-expansion about the other centre).

Sphere 1 (radius a1) at z = 0, sphere 2 (radius a2) at z = d. Generalized velocities
q = (R1', R2', U1, U2): radial wall velocities and translational velocities along +z.
Kinetic energy T = 1/2 q' M(d) q'. Time-averaged force on sphere 2 along +z (second order):
F2 = 1/4 Re(q^H dM/dd q) for complex amplitudes q (equal and opposite on sphere 1).
"""

import numpy as np
from scipy.special import gammaln

RHO = 1000.0


def added_mass(a1, a2, d, order=120):
    n = np.arange(order + 1)
    m = n[:, None]
    nn = n[None, :]
    # log of C(m+n, n) (a1/d)^m (a2/d)^n, then signs
    logc = gammaln(m + nn + 1) - gammaln(m + 1) - gammaln(nn + 1)
    # sphere 1 equations: -(m+1) A_m + m a1^(m-1) sum_n B_n a2^(n+2) (-1)^n C d^-(m+n+1)
    # unknowns scaled: A_n, B_n as given (velocity units)
    k1 = m * np.exp(logc + m * np.log(a1 / d) + nn * np.log(a2 / d)) * (a2 * a2 / (a1 * d)) * (-1.0) ** nn
    k2 = m * np.exp(logc + m * np.log(a2 / d) + nn * np.log(a1 / d)) * (a1 * a1 / (a2 * d)) * (-1.0) ** m
    N = order + 1
    S = np.zeros((2 * N, 2 * N))
    S[:N, :N] = np.diag(-(n + 1.0))
    S[:N, N:] = k1
    S[N:, N:] = np.diag(-(n + 1.0))
    S[N:, :N] = k2
    rhs = np.zeros((2 * N, 4))
    rhs[0, 0] = 1.0       # R1'
    rhs[N, 1] = 1.0       # R2'
    rhs[1, 2] = 1.0       # U1
    rhs[N + 1, 3] = 1.0   # U2
    sol = np.linalg.solve(S, rhs)
    A, B = sol[:N], sol[N:]
    # surface potential coefficients c_0, c_1 on each sphere
    def outer(m_, a_self, a_other, coef_other, sign_rule):
        logc_ = gammaln(m_ + n + 1) - gammaln(m_ + 1) - gammaln(n + 1)
        w = np.exp(logc_ + n * np.log(a_other / d) + m_ * np.log(a_self / d)) * (a_other ** 2 / d)
        s = (-1.0) ** n if sign_rule == 1 else (-1.0) ** m_ * np.ones_like(n, dtype=float)
        return (w * s) @ coef_other
    c10 = A[0] * a1 + outer(0, a1, a2, B, 1)
    c11 = A[1] * a1 + outer(1, a1, a2, B, 1)
    c20 = B[0] * a2 + outer(0, a2, a1, A, 2)
    c21 = B[1] * a2 + outer(1, a2, a1, A, 2)
    w = np.array([4 * np.pi * a1 ** 2, 4 * np.pi * a2 ** 2, 4 * np.pi * a1 ** 2 / 3, 4 * np.pi * a2 ** 2 / 3])
    C = np.vstack([c10, c20, c11, c21])
    return -RHO * w[:, None] * C


def dM(a1, a2, d, order=120, rel=1e-5):
    h = rel * d
    return (added_mass(a1, a2, d + h, order) - added_mass(a1, a2, d - h, order)) / (2 * h)


def mean_force(a1, a2, d, R1dot, R2dot, translate, order=120):
    """F2 (+z, on sphere 2) for complex radial wall velocity amplitudes. translate: massless
    spheres free to translate (inviscid, momentum of each sphere's translational mode zero)."""
    M = added_mass(a1, a2, d, order)
    q = np.array([R1dot, R2dot, 0, 0], dtype=complex)
    if translate:
        q[2:] = np.linalg.solve(M[2:, 2:], -M[2:, :2] @ q[:2])
    return 0.25 * np.real(np.conj(q) @ dM(a1, a2, d, order) @ q), q


def monopole_force(a1, a2, d, R1dot, R2dot):
    return -4 * np.pi * RHO * a1 ** 2 * a2 ** 2 / d ** 2 * 0.5 * np.real(R1dot * np.conj(R2dot))
