"""5 ms against 2.5 ms steps for the complex changes (2026-11-09; from the sheet: the two with steps above 0.1 R; from
the three-dimensional cluster: all five, whose first steps reach 0.9 R).

usage (repository root):  python step_check_complex.py ANALYSIS_JSON ANALYSIS_NPZ DATA_DIR OUT_JSON OUT_PNG
ANALYSIS_JSON / _NPZ: analyse_complex.py with TAG step_check over the runs (each change at 5 ms and at 2.5 ms, the
2.5 ms run named STEP_<name>_dt2.5ms). Per pair: the largest difference of the share on the zero set
over the saved frames, the summary numbers side by side, the root-mean-square distance between the same bubbles
at the end, and the steps above 0.1 R (R = 1.78 um) of each run.
"""

import json
import shutil
import sys
import tempfile
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
from matplotlib import font_manager

INK, MUTED, GRID = "#1f2328", "#5b6470", "#e6e8eb"
FIVE, HALF = "#2a78d6", "#eb6834"
R = 1.78e-6
KEYS = ("share_min", "share_mean_during_change", "share_end", "share_end_of_crossing_hold", "at_crossing_um", "end_um",
        "bend_amplitude_ratio", "bend_phase_lag_deg", "bend_lag_s", "pieces_max", "direction_from_s",
        "lag_deg_max_with_direction", "lag_deg_mean_during_change_with_direction")
TITLES = dict(spin="Spin 180°", pivot="Turn 60° about one end", line_cross="Line → cross → line", flap="Flap")


def main():
    summary = {row["name"]: row for row in json.loads(Path(sys.argv[1]).read_text())}
    series, data = np.load(sys.argv[2]), Path(sys.argv[3])
    pairs = [(name, f"STEP_{name}_dt2.5ms") for name in summary if not name.startswith("STEP_")]
    result = []
    for coarse, fine in pairs:
        steps = {}
        for name in (coarse, fine):
            c = np.load(data / name / "compact.npz")
            displacement = c["step_max_displacement_m"]
            steps[name] = dict(share_above_0_1R=float(np.mean(displacement > 0.1 * R)),
                               max_over_R=float(displacement.max() / R))
        end = [np.load(data / name / "compact.npz")["positions_m"][-1] for name in (coarse, fine)]
        result.append(dict(
            coarse=coarse, fine=fine,
            share_max_difference=float(np.abs(series[f"{coarse}__share_on_zero_set"]
                                              - series[f"{fine}__share_on_zero_set"]).max()),
            summary={key: [summary[coarse][key], summary[fine][key]] for key in KEYS if key in summary[coarse]},
            end_rms_distance_um=float(np.sqrt(np.mean(np.sum((end[0] - end[1]) ** 2, axis=1))) * 1e6),
            steps=steps))
        print(json.dumps(result[-1]))
    Path(sys.argv[4]).write_text(json.dumps(result, indent=1) + "\n")

    with tempfile.TemporaryDirectory() as font_dir:
        font_manager.fontManager.addfont(str(shutil.copy("/mnt/c/Windows/Fonts/arial.ttf", Path(font_dir) / "arial.ttf")))
        plt.rcParams.update({"font.family": "Arial", "font.size": 9, "axes.edgecolor": MUTED, "axes.labelcolor": INK,
                             "xtick.color": MUTED, "ytick.color": MUTED})
        fig, axes = plt.subplots(2, len(pairs), figsize=(4.2 * len(pairs), 5.6), dpi=150, squeeze=False)
        fig.subplots_adjust(left=0.09, right=0.98, bottom=0.09, top=0.88, wspace=0.25, hspace=0.45)
        for c, (coarse, fine) in enumerate(pairs):
            form = summary[coarse]["form"]
            for name, colour, label in ((coarse, FIVE, "5 ms step"), (fine, HALF, "2.5 ms step")):
                t = series[f"{name}__t_s"]
                axes[0, c].plot(t, series[f"{name}__share_on_zero_set"], color=colour, lw=1.3, label=label)
                if form == "flap":
                    axes[1, c].plot(t, series[f"{name}__bend_sheet_per_mm"], color=colour, lw=1.3)
                else:
                    axes[1, c].plot(t, series[f"{name}__length_um"], color=colour, lw=1.3)
                    axes[1, c].plot(t, series[f"{name}__width_um"], color=colour, lw=1.3, ls="--")
            title = TITLES[form] + (f", period {summary[coarse]['period_s']:g} s" if form == "flap" else "")
            axes[0, c].set_title(title, loc="left", fontsize=9.5, color=INK)
            axes[0, c].set_ylabel(f"Share within {R * 1e6:.2f} µm\nof the zero line")
            axes[0, c].set_ylim(0, 1.05)
            axes[1, c].set_ylabel("Bend of the bubbles b (1/mm)" if form == "flap"
                                  else "Section length (solid), width (dashed) (µm)")
            axes[1, c].set_xlabel("Time (s)")
            for ax in axes[:, c]:
                ax.grid(color=GRID, lw=0.6)
                ax.set_axisbelow(True)
                for side in ("top", "right"):
                    ax.spines[side].set_visible(False)
        handles, labels = axes[0, 0].get_legend_handles_labels()
        fig.legend(handles, labels, loc="upper right", ncol=2, frameon=False, fontsize=9)
        fig.savefig(sys.argv[5])


main()
