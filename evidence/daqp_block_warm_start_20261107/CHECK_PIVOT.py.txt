"""Block warm start (2026-11-07): the pivoted path for guessed contacts that depend on the others.

usage (repository root):  python check_pivot.py OLD_LIBRARY NEW_TIMED_LIBRARY OUT_JSON
OLD_LIBRARY: the previous version (DAQP takes the guessed constraints in one by one);
NEW_TIMED_LIBRARY: the block version with counters (slot 28 pivoted loads, 29 plain loads,
30 guessed constraints left out).
1. Least-distance problems with dependent rows (duplicates and sums of other rows), the guess
   marking the dependent rows as well: same w as a cold start and as the old version.
2. The contact steps of tests/test_pretrain.py::test_compiled_contact_step_follows_the_python_step
   (formed clusters pressed together, three nearly touching bubbles, the cycling network), run
   with both libraries: same positions, gates and counts; how often the pivoted path was taken.
"""

import ctypes
import json
import sys
from pathlib import Path

import numpy as np

from microbubble.contact import CompiledContactStep
from microbubble.forces import CPUForces, adhesion_forces
from microbubble.initial_conditions import load_state
from workflows.f10ag import carried_apecss_response, load_pretrain_policy, pretrain_context

ROOT = Path(".")
old_path, new_path, out_json = sys.argv[1], sys.argv[2], Path(sys.argv[3])
old, new = ctypes.CDLL(old_path), ctypes.CDLL(new_path)
P = lambda a, t=ctypes.c_double: a.ctypes.data_as(ctypes.POINTER(t))


def counters():
    c = np.zeros(32)
    new.contact_bill_get(P(c))
    return dict(pivoted=int(c[28]), plain=int(c[29]), left_out=int(c[30]))


def solve(lib, M, d, guess):
    m, n = M.shape
    w, lam = np.zeros(n), np.zeros(m)
    active, it = np.zeros(m, np.int32), ctypes.c_int(10000)
    g = np.ascontiguousarray(guess, np.int32)
    status = lib.contact_ldp(n, m, P(np.ascontiguousarray(M)), P(d), P(g, ctypes.c_int),
                             ctypes.byref(it), P(w), P(lam), P(active, ctypes.c_int))
    return dict(status=status, w=w, lam=lam, active=active.astype(bool), iterations=it.value)


synthetic = []
rng = np.random.default_rng(20261107)
for case in range(20):
    n, base = 40, 30
    M = rng.normal(size=(base, n))
    d = rng.uniform(0.2, 1.0, base)  # w = 0 violates every row
    extra_rows, extra_d = [], []
    for _ in range(8):
        i, j = rng.choice(base, 2, replace=False)
        if rng.random() < 0.5:  # duplicate (a contact listed twice)
            extra_rows.append(M[i])
            extra_d.append(d[i])
        else:  # positive sum of two rows: active whenever both are
            a, b = rng.uniform(0.5, 2.0, 2)
            extra_rows.append(a * M[i] + b * M[j])
            extra_d.append(a * d[i] + b * d[j])
    M, d = np.vstack([M, extra_rows]), np.concatenate([d, extra_d])
    cold = solve(old, M, d, np.zeros(len(d)))
    residual = M @ cold["w"] - d
    guess = cold["active"] | (np.abs(residual) < 1e-9)  # the dependent rows active at w too
    new.contact_bill_reset()
    a, b = solve(old, M, d, guess), solve(new, M, d, guess)
    synthetic.append(dict(
        case=case, rows=len(d), guessed=int(guess.sum()), counters=counters(),
        dependent=int(guess.sum() - np.linalg.matrix_rank(M[guess])),
        status_old=a["status"], status_new=b["status"],
        iterations_old=a["iterations"], iterations_new=b["iterations"],
        w_difference_new_vs_cold=float(np.abs(b["w"] - cold["w"]).max()),
        w_difference_new_vs_old=float(np.abs(b["w"] - a["w"]).max()),
        stationarity_new=float(np.abs(M.T @ b["lam"] - b["w"]).max()),
        min_multiplier_new=float(b["lam"].min()),
        max_violation_new=float(max(0.0, (d - M @ b["w"]).max())),
    ))
    print(f"synthetic {case:2d}: guessed {guess.sum()} of {len(d)}, dependent {synthetic[-1]['dependent']}, {counters()}, "
          f"|w_new - w_cold| {synthetic[-1]['w_difference_new_vs_cold']:.1e}, "
          f"iterations old {a['iterations']} new {b['iterations']}")

# 2. the contact steps of the test (same cases, same order, each library carries its own warm start)
c = pretrain_context(ROOT, 2.5e6)
policy = load_pretrain_policy(ROOT)
carried, _ = carried_apecss_response(ROOT, {}, policy, 2.5e6, 1, Path(out_json).parent / "apecss_tmp")
R = c.physics.radius_m
zeta = 6 * np.pi * c.physics.viscosity_Pa_s * R


class Fresh:
    def __init__(self, carried):
        self.carried, self.knots = carried, carried.knots

    def query(self, p):
        return self.carried.probe(p)

    def probe(self, p):
        return self.carried.probe(p)


c.response = Fresh(carried)
f = CPUForces(c.response, c.physics, c.discs, "pairwise")
f.control([0, 0, 3000, 0], [0, 0, 0, 0])
adhesion = policy["adhesion"]["coefficient_N_per_Pa2"] * 6000.0**2
cases = []
for name, steps in (("FORMED_N32_a1.0_s1", (1e-5, 1e-4, 1e-3)), ("FORMED_N128_a1.0_s0", (1e-5, 1e-4))):
    x = load_state(ROOT / "inputs/pretrain_formed_states.h5", name)[0]
    F = f(x)["total_force_N"] + adhesion_forces(x, R, adhesion, policy["adhesion"]["range_m"])
    cases.append((name, x, F, steps))
x = np.array([[0, 0, 0], [2.001 * R, 0, 0], [4.003 * R, 0.01 * R, 0]]) + [20e-6, 0, 0]
for push in (1e-8, 1e-7):
    cases.append((f"three bubbles, push {push:g} N", x, np.array([[push, 0, 0], [0, 0, 0], [-push, 0, 0]]), (1e-6,)))
x = np.load(ROOT / "evidence/pretrain_tier_20261102/ACTIVE_SET_CYCLE_STATE.npy")
cases.append(("cycling network", x, f(x)["total_force_N"], (1e-6,)))
carried.close()
step_old, step_new = CompiledContactStep(old_path, policy), CompiledContactStep(new_path, policy)
steps = []
for name, x, F, dts in cases:
    for dt in dts:
        new.contact_bill_reset()
        it_old, it_new = step_old.iterations, step_new.iterations
        a, b = step_old(x, F, R, zeta, dt), step_new(x, F, R, zeta, dt)
        same = {k: a[k] == b[k] for k in ("gates", "converged", "solver_status", "active_count", "candidate_count")}
        steps.append(dict(case=name, bubbles=len(x), dt_s=dt, counters=counters(), same=same,
                          iterations_old=step_old.iterations - it_old, iterations_new=step_new.iterations - it_new,
                          active_count=a["active_count"],
                          max_position_difference_m=float(np.abs(a["positions_m"] - b["positions_m"]).max())))
        print(f"{name:32s} dt {dt:g}: {counters()}, same {all(same.values())}, active {a['active_count']}, "
              f"iterations old {steps[-1]['iterations_old']} new {steps[-1]['iterations_new']}, "
              f"|dx| {steps[-1]['max_position_difference_m']:.1e} m")
out_json.write_text(json.dumps(dict(synthetic=synthetic, contact_steps=steps), indent=1) + "\n")
