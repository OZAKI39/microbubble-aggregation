"""Re-run prototype cases (displacement controller, 3% R) with the repository implementation
and compare the saved positions and solve counts with the prototype outputs.

usage: repro.py REPO_ROOT PROTO_OUT_DIR WORK_DIR
"""
import sys, json, time
from pathlib import Path
import numpy as np, h5py

root, proto_out, work = Path(sys.argv[1]), Path(sys.argv[2]), Path(sys.argv[3])
sys.path.insert(0, str(root))
from workflows.f10ag import context, load_pretrain_policy
from microbubble.control import pretrain_segments
from microbubble.integration import integrate_pretrain
from microbubble.initial_conditions import load_state
from microbubble.io import identity

CASES = {
    "N1_cluster32_1000": dict(state="AG32_local", amp=1000, offset=[40e-6, 0, 0], T=0.005, save=0.00025, coupling="collective"),
    "N2_compact64_1000": dict(state="N64_compact_reference", amp=1000, offset=[40e-6, 0, 0], T=0.002, save=0.0001, coupling="collective"),
    "N3_cluster32_3000": dict(state="AG32_local", amp=3000, offset=[20e-6, 0, 0], T=0.005, save=0.00025, coupling="collective"),
    "N4_settled32_3000": dict(init="E1_cluster32_1000__ref_pre_0p5us", amp=3000, offset=[0, 0, 0], T=0.005, save=0.00025, coupling="collective"),
    "N4L_settled32_3000": dict(init="E1_cluster32_1000__ref_pre_0p5us", amp=3000, offset=[0, 0, 0], T=0.020, save=0.0005, coupling="collective"),
    "N5_loose64_3000": dict(state="N64_loose_r0", amp=3000, offset=[40e-6, 0, 0], T=0.010, save=0.0005, coupling="collective"),
    "E1_cluster32_1000": dict(state="AG32_local", amp=1000, offset=[40e-6, 0, 0], T=0.005, save=0.00025, coupling="pairwise"),
    "E2_compact64_1000": dict(state="N64_compact_reference", amp=1000, offset=[40e-6, 0, 0], T=0.002, save=0.0001, coupling="pairwise"),
    "E3_cluster32_3000": dict(state="AG32_local", amp=3000, offset=[20e-6, 0, 0], T=0.005, save=0.00025, coupling="pairwise"),
    "E4_settled32_3000": dict(init="E1_cluster32_1000__ref_pre_0p5us", amp=3000, offset=[0, 0, 0], T=0.005, save=0.00025, coupling="pairwise"),
}
c = context(root, device="cpu")
policy = load_pretrain_policy(root)
ident = identity(root)
rows = {}
for case, k in CASES.items():
    if "init" in k:
        x0 = np.load(proto_out / f"{k['init']}.npz")["X"][-1]
    else:
        x0, _ = load_state(root / "inputs/initial_states.h5", k["state"])
    task = dict(id=case, duration=k["T"], coupling=k["coupling"], save_interval_s=k["save"], offset_m=k["offset"],
                amplitudes=[0, 0, k["amp"], k["amp"]], phases=[0, 0, 0, 0])
    seg = pretrain_segments(task, policy["max_source_amplitude_Pa_equivalent"], policy["max_segments"])
    t0 = time.perf_counter()
    r = integrate_pretrain(task, c, x0, seg, policy, work / case, ident)
    wall = time.perf_counter() - t0
    with h5py.File(work / case / "trajectory.h5") as h:
        t, X = h["time_s"][:], h["positions_m"][:]
    proto = np.load(proto_out / f"{case}__dis_pre_3pct.npz")
    pj = json.load(open(proto_out / f"{case}__dis_pre_3pct.json"))
    same_times = len(t) == len(proto["times"]) and float(np.abs(t - proto["times"]).max()) < 1e-15
    rows[case] = dict(
        coupling=k["coupling"], status=r["status"], completed_s=r["completed_s"],
        repo_solves=r["constrained_solves"], prototype_solves=pj["constrained_solves"],
        same_save_times=same_times,
        max_position_difference_m=float(np.linalg.norm(X - proto["X"], axis=2).max()) if same_times else None,
        repo_min_singular_value=r["min_collective_singular_value"], prototype_min_singular_value=pj.get("min_singular_value"),
        repo_wall_s=round(wall, 2), prototype_wall_s=round(pj["wall_s"], 2),
        final_aspect=r["final"]["xy_f10y"]["aspect"], final_contacts=r["contacts_final"],
    )
    print(case, rows[case], flush=True)
out = dict(
    purpose="Repository implementation vs prototype (displacement controller, 3% R, pretraining tier), same cases and save times",
    environment=dict(python=sys.version.split()[0], versions=ident["versions"]),
    prototype_environment="Python 3.13, NumPy 2.5.3, SciPy 1.18.1, OSQP 1.1.3 (session scratch prototype, see prototype/*.py.txt)",
    not_rerun="E4L (20 ms pairwise, 248 s) and E5 (pairwise loose cloud, budget-capped) were not repeated here",
    rows=rows,
)
json.dump(out, open(work / "REPO_REPRODUCTION.json", "w"), indent=1, default=str)
print("REPRO_DONE")
