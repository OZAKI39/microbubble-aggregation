"""Animation: the bursting run and the two controls side by side (2026-11-08).

usage (repository root):  PYTHONPATH=. python animation_burst.py RUNS_DIR OUT_MP4
The 64-bubble start with random actions at 2-3 MHz and adhesion 1e-12 N, 0.25 ms steps: drawn radii with
collective coupling (bursts), equal radii, pairwise coupling. Every saved frame (10 ms) at 20 frames/s (1 s of
simulation takes 5 s). Top view centred on the largest piece (the view follows it), +-60 um; colour = radius
(resonant radius at 2.82 MHz: 2.0 um). Text: time, pieces, the share the largest piece keeps, how far it moved.
"""

import shutil
import sys
import tempfile
from pathlib import Path

import h5py
import imageio_ffmpeg
import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
from matplotlib import animation, font_manager
from matplotlib.collections import EllipseCollection
from scipy.sparse import coo_matrix
from scipy.sparse.csgraph import connected_components
from scipy.spatial import cKDTree

ARIAL_DIR = Path("/mnt/c/Windows/Fonts")
INK, MUTED = "#1f2328", "#5b6470"
R_REF = 1.78e-6
HALF_UM = 60
RUNS = [("STEPFINE_N64_random_adh1e-12_dt0.25ms", "drawn radii, collective coupling"),
        ("BURST_N64_random_adh1e-12_dt0.25ms_equal_sizes", "all radii 1.78 um, collective coupling"),
        ("BURST_N64_random_adh1e-12_dt0.25ms_pairwise", "drawn radii, pairwise coupling")]


def piece_labels(x, radii):
    pairs = np.array(sorted(cKDTree(x).query_pairs(2 * radii.max() + 0.5 * R_REF)), dtype=int).reshape(-1, 2)
    d = np.linalg.norm(x[pairs[:, 1]] - x[pairs[:, 0]], axis=1)
    pairs = pairs[d - radii[pairs[:, 0]] - radii[pairs[:, 1]] < 0.5 * R_REF]
    return connected_components(coo_matrix((np.ones(len(pairs)), (pairs[:, 0], pairs[:, 1])), shape=(len(x), len(x))),
                                directed=False)


def main():
    runs, out = Path(sys.argv[1]), Path(sys.argv[2])
    with tempfile.TemporaryDirectory() as font_dir:
        for font in ("arial.ttf", "arialbd.ttf"):
            if (ARIAL_DIR / font).is_file():
                font_manager.fontManager.addfont(str(shutil.copy(ARIAL_DIR / font, Path(font_dir) / font)))
        plt.rcParams.update({"font.family": "Arial", "font.size": 9, "axes.edgecolor": MUTED,
                             "xtick.color": MUTED, "ytick.color": MUTED,
                             "animation.ffmpeg_path": imageio_ffmpeg.get_ffmpeg_exe()})
        fig, axes = plt.subplots(1, 3, figsize=(13.2, 5.0), dpi=100)
        fig.subplots_adjust(left=0.05, right=0.92, bottom=0.11, top=0.8, wspace=0.2)
        fig.suptitle("64 bubbles, random actions at 2-3 MHz, adhesion 1e-12 N, 0.25 ms steps (1 s takes 5 s); view follows the "
                     "largest piece", x=0.05, ha="left", fontsize=10, color=INK)
        state = []
        for ax, (name, label) in zip(axes, RUNS):
            with h5py.File(runs / name / "trajectory.h5") as h:
                t, x, radii = h["time_s"][()], h["positions_m"][()], h["radii_m"][()]
            dots = EllipseCollection(2 * radii * 1e6, 2 * radii * 1e6, 0, units="xy", offsets=np.zeros((len(radii), 2)),
                                     offset_transform=ax.transData, cmap="coolwarm", edgecolors="white", linewidths=0.15)
            dots.set_array(radii * 1e6)
            dots.set_clim(0.8, 3.0)
            ax.add_collection(dots)
            text = ax.text(0.02, 0.98, "", transform=ax.transAxes, fontsize=8, color=INK, va="top",
                           bbox=dict(facecolor="white", edgecolor="none", alpha=0.85, pad=1.5))
            ax.set_title(label, loc="left", fontsize=9, color=INK)
            ax.set_xlim(-HALF_UM, HALF_UM)
            ax.set_ylim(-HALF_UM, HALF_UM)
            ax.set_aspect("equal")
            ax.set_xlabel("x - largest piece (um)")
            state.append(dict(t=t, x=x, radii=radii, dots=dots, text=text, start=x[0].mean(axis=0)))
        axes[0].set_ylabel("y - largest piece (um)")
        bar = fig.colorbar(state[0]["dots"], cax=fig.add_axes([0.935, 0.15, 0.012, 0.6]))
        bar.set_label("radius (um)")

        def draw(i):
            for s in state:
                j = min(i, len(s["t"]) - 1)
                x = s["x"][j]
                n, labels = piece_labels(x, s["radii"])
                main = labels == np.argmax(np.bincount(labels))
                centre = x[main].mean(axis=0)
                s["dots"].set_offsets((x - centre)[:, :2] * 1e6)
                s["text"].set_text(f"t = {s['t'][j]:.2f} s, pieces {n}\nlargest keeps {main.mean():.0%}, moved "
                                   f"{np.linalg.norm(centre - s['start']) * 1e6:.0f} um")
            return []

        frames = max(len(s["t"]) for s in state)
        writer = animation.FFMpegWriter(fps=20, codec="libx264", extra_args=["-pix_fmt", "yuv420p"])
        animation.FuncAnimation(fig, draw, frames=frames, blit=False).save(out, writer=writer)
        plt.close(fig)


main()
