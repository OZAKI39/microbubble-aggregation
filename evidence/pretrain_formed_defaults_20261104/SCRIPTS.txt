# ===== final/analyse_final.py =====
"""Final A/B comparison of the new pretraining defaults (adhesion by ratio, delay, formed clusters).

usage (repository root): PYTHONPATH=.:visualization python analyse_final.py RUNS_DIR MB_RESULTS OUT.json FIG.png
Uses the functions of visualization/ab_protocol_figures.py (same windows, rasterization, metrics).
References: the branch's protocol and random-action runs (old defaults, AG32_local) and A/B.
"""
import json
import sys
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np

from ab_protocol_figures import (PROTOCOL_RUNS, RANDOM_RUNS, deformation_stats, frame_shapes, load_run,
                                 quantiles, real_cluster_windows, real_windows, sim_windows, speed_px_s)
from microbubble.parameters import Physics

runs_dir, mb, out, fig_path = Path(sys.argv[1]), Path(sys.argv[2]), Path(sys.argv[3]), Path(sys.argv[4])
R = Physics().radius_m


def windows_of(result, config):
    run = load_run(result, config)
    return sim_windows(run, frame_shapes(run, R), R), run


def summary(ws):
    d = deformation_stats(ws)
    return dict(windows=d["windows"], split_share=d["split_share"],
                per_px=d["dln_aspect_per_px_first_1s"].get("median"),
                dln_first_1s=d["dln_aspect_first_1s"].get("median"), aspect=d["aspect"].get("median"),
                aspect_p90=d["aspect"].get("p90"),
                speed_first_0p5s=quantiles([speed_px_s(w) for w in ws if w["t_since_s"] < 0.5]).get("median"),
                speed_after_1s=quantiles([speed_px_s(w) for w in ws if w["t_since_s"] >= 1.0]).get("median"))


groups = {}
new = {}
for d in sorted(runs_dir.glob("*_formed*")):
    if not (d / "SUMMARY.json").is_file():
        continue
    cfg = json.loads((d / "EXECUTION.json").read_text())["config"]
    (d / "config_used.json").write_text(json.dumps(cfg))
    ws, run = windows_of(d, d / "config_used.json")
    s = run["summary"]
    new[d.name] = dict(summary(ws), status=s["status"], wall_s=s["wall_s"], adhesion_force_N=s["task"]["adhesion_force_N"])
    key = ("new protocol N128" if "128" in d.name else "new protocol N32") if d.name.startswith("protocol") else "new random actions N32"
    groups.setdefault(key, []).extend(ws)
for name, (result, config) in PROTOCOL_RUNS.items():
    ws, _ = windows_of(Path(result), Path(config))
    groups.setdefault("old protocol " + name.split("_")[1], []).extend(ws)
for name, (result, config) in RANDOM_RUNS.items():
    ws, _ = windows_of(Path(result), Path(config))
    groups.setdefault("old random actions (collective)", []).extend(ws)
real = {e: real_windows(mb / d) for e, d in (("A", "A_offline"), ("B", "B_offline"))}
real_on = [w for w in real_cluster_windows(real) if w["action"] != "none"]
table = {k: summary(v) for k, v in groups.items()}
table["real A/B (PZT on)"] = summary(real_on)
for k, v in table.items():
    print(f"{k:34s} per-px {v['per_px']}  dln/0.1s {v['dln_first_1s']}  aspect {v['aspect']} (p90 {v['aspect_p90']})  split {v['split_share']}  "
          f"speed {v['speed_first_0p5s']} -> {v['speed_after_1s']}")
for k, v in new.items():
    print(f"  {k:32s} per-px {v['per_px']}  aspect {v['aspect']}  split {v['split_share']}  speed {v['speed_first_0p5s']}  wall {v['wall_s']:.0f}s  {v['status']}  F_adh {v['adhesion_force_N']:.2e}")
out.write_text(json.dumps(dict(note=__doc__, groups=table, runs=new), indent=1, default=float) + "\n")

INK, INK2, MUTED, GRID, SURF = "#0b0b0b", "#52514e", "#8a8984", "#e6e5e1", "#fcfcfb"
order = [("real A/B (PZT on)", "#eda100", "real A/B\n(PZT on)"),
         ("old protocol pairwise", MUTED, "old\npairwise"),
         ("old protocol collective", "#eb6834", "old\ncollective"),
         ("old random actions (collective)", "#f4a582", "old\ncollective\nrandom"),
         ("new protocol N32", "#2a78d6", "new\nN32"),
         ("new protocol N128", "#1c5cab", "new\nN128"),
         ("new random actions N32", "#86b6ef", "new\nN32\nrandom")]
plt.rcParams.update({"font.family": "DejaVu Sans", "font.size": 9, "axes.edgecolor": MUTED,
                     "axes.labelcolor": INK2, "xtick.color": INK2, "ytick.color": INK2})
fig, axes = plt.subplots(1, 3, figsize=(13.2, 4.0))
fig.patch.set_facecolor(SURF)
spec = (("per_px", "a  shape change per pixel moved", "median |d ln aspect| per px", True),
        ("split_share", "b  windows in which the cluster splits", "share of 0.1 s windows", False),
        ("aspect", "c  aspect ratio of the cluster", "median aspect ratio", False))
for ax, (key, title, ylabel, log) in zip(axes, spec):
    ax.set_facecolor(SURF)
    for s_ in ("top", "right"):
        ax.spines[s_].set_visible(False)
    for i, (g, color, label) in enumerate(order):
        v = table.get(g, {}).get(key)
        if v is None:
            ax.text(i, 0.01 if not log else 2e-4, "n/a", ha="center", fontsize=7, color=INK2)
            continue
        ax.bar(i, v, 0.62, color=color)
    ax.set_xticks(range(len(order)))
    ax.set_xticklabels([o[2] for o in order], fontsize=6.6)
    ax.set_title(title, loc="left", color=INK, fontsize=9.5)
    ax.set_ylabel(ylabel)
    ax.grid(True, axis="y", which="both", color=GRID, lw=0.5)
    ax.set_axisbelow(True)
    if log:
        ax.set_yscale("log")
    ref = table["real A/B (PZT on)"].get(key)
    if ref is not None:
        ax.axhline(ref, color="#eda100", lw=1, ls=":")
axes[2].set_ylim(1, 1.9)
fig.text(0.01, 0.005, "old = branch runs (AG32_local grid start, no delay, no adhesion); new = default since 2026-11-04 (delay, adhesion k x A_ref^2, formed clusters). "
         "Protocol X- -> Y- -> X-; random = 0.1 s actions. Exploratory, 0.6 um/px provisional.", fontsize=6.6, color=INK2)
fig.tight_layout(rect=(0, 0.05, 1, 1))
fig.savefig(fig_path, dpi=200, facecolor=SURF)

# ===== adh/analyse_runs.py =====
"""Softness metrics (functions of visualization/ab_protocol_figures.py) for any set of PRETRAIN runs.

usage (repository root): PYTHONPATH=.:visualization python analyse_runs.py OUT.json RUN_DIR [RUN_DIR ...]
Each RUN_DIR needs trajectory.h5, SUMMARY.json and EXECUTION.json (its config is used).
"""
import json
import sys
from pathlib import Path

import numpy as np

from ab_protocol_figures import (deformation_stats, frame_shapes, load_run, merged_from, quantiles,
                                 sim_windows, speed_px_s)
from microbubble.parameters import Physics

R = Physics().radius_m
rows = {}
for d in map(Path, sys.argv[2:]):
    cfg = json.loads((d / "EXECUTION.json").read_text())["config"]
    (d / "config_used.json").write_text(json.dumps(cfg))
    run = load_run(d, d / "config_used.json")
    shapes = frame_shapes(run, R)
    w = sim_windows(run, shapes, R)
    s = deformation_stats(w)
    t = run["times_s"]
    rows[d.name] = dict(
        amplitude_Pa_eq=max(cfg["control"]["segments"][0]["amplitudes"]),
        adhesion_force_N=cfg.get("adhesion_force_N", 0.0),
        split_share=s["split_share"],
        dln_aspect_per_px_first_1s=s["dln_aspect_per_px_first_1s"],
        dln_aspect_first_1s=s["dln_aspect_first_1s"],
        aspect=s["aspect"],
        speed_px_s=quantiles([speed_px_s(x) for x in w]),
        aspect_over_time=[round(shapes[int(np.searchsorted(t, v))]["aspect"], 3) for v in (0.0, 0.25, 0.5, 1.0, t[-1])],
        pieces_end=int(shapes[-1]["pieces"]),
        wall_s=run["summary"]["wall_s"],
        status=run["summary"]["status"],
    )
    r = rows[d.name]
    print(f"{d.name:42s} split {r['split_share']:.2f}  per-px med {r['dln_aspect_per_px_first_1s'].get('median')}  (n {r['dln_aspect_per_px_first_1s'].get('n')})  "
          f"dln/0.1s {r['dln_aspect_first_1s'].get('median')}  aspect {r['aspect']['median']:.2f}  speed {r['speed_px_s'].get('median'):.0f} px/s  "
          f"pieces {r['pieces_end']}  aspect(t) {r['aspect_over_time']}", flush=True)
Path(sys.argv[1]).write_text(json.dumps(dict(note=__doc__, rows=rows), indent=1, default=float) + "\n")
