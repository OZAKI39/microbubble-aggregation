===== scratch/adh2/analyse_adh2.py
"""Matched-speed adhesion check: shape change of each run against its own rigid-body floor and A/B.

For each run: 0.1 s windows of visualization/ab_protocol_figures.py (0.6 um/px), median shape
change per window and per px moved, aspect, split share, speed. The floor: the initial cluster
moved rigidly along the run's own centroid path, rasterised the same way (what a cluster that
does not deform at all would give). A/B: cluster windows with the PZT on in the first 1 s,
binned by speed; the 120-250 px/s bin matches the simulated speed.
usage (repo root): PYTHONPATH=.:visualization python analyse_adh2.py MB_RESULTS OUT.json RUN_DIR ...
"""
import json
import sys
from pathlib import Path

import numpy as np
from ab_protocol_figures import (deformation_stats, frame_shapes, load_run, real_cluster_windows,
                                 real_windows, sim_windows, speed_px_s)

from microbubble.parameters import Physics

R = Physics().radius_m


def stats(ws):
    d = deformation_stats(ws)
    return dict(dln=d["dln_aspect_first_1s"].get("median"), per_px=d["dln_aspect_per_px_first_1s"].get("median"),
                aspect=d["aspect"].get("median"), aspect_p90=d["aspect"].get("p90"), split=d["split_share"],
                speed=float(np.median([speed_px_s(w) for w in ws])))


mb, out = Path(sys.argv[1]), Path(sys.argv[2])
real = {e: real_windows(mb / d) for e, d in (("A", "A_offline"), ("B", "B_offline"))}
on = [w for w in real_cluster_windows(real) if w["action"] != "none" and w["t_since_s"] < 1.0]
band = [w for w in on if 120 <= speed_px_s(w) < 250]
rows = dict(real_120_250=dict(n=len(band), dln=float(np.median([w["dln_aspect"] for w in band])),
                              per_px=float(np.median([w["dln_aspect"] / w["disp_px"] for w in band])),
                              aspect=float(np.median([w["aspect"] for w in band])),
                              aspect_p90=float(np.percentile([w["aspect"] for w in band], 90))))
print("real 120-250 px/s", rows["real_120_250"])
for d in map(Path, sys.argv[3:]):
    cfg = json.loads((d / "EXECUTION.json").read_text())["config"]
    (d / "config_used.json").write_text(json.dumps(cfg))
    run = load_run(d, d / "config_used.json")
    s = stats(sim_windows(run, frame_shapes(run, R), R))
    x0 = run["positions_m"][0]
    centre = run["positions_m"].mean(1)
    rigid = dict(run, positions_m=np.array([x0 - x0.mean(0) + c for c in centre]))
    floor = stats(sim_windows(rigid, frame_shapes(rigid, R), R))
    summ = run["summary"]
    rows[d.name] = dict(sim=s, rigid_floor=floor, status=summ["status"], wall_s=round(summ["wall_s"]),
                        adhesion_force_N=summ["task"]["adhesion_force_N"])
    print(f"{d.name:10s} {summ['status']} {summ['wall_s']:5.0f}s speed {s['speed']:5.0f} dln {s['dln']:.4f} (floor {floor['dln']:.4f}) "
          f"per-px {s['per_px']:.5f} (floor {floor['per_px']:.5f}) aspect {s['aspect']:.3f} p90 {s['aspect_p90']:.3f} split {s['split']:.2f}")
out.write_text(json.dumps(rows, indent=1) + "\n")
===== scratch/adh2/noise_floor.py
"""Pixel floor of the shape metric: the initial formed N32 cluster moved rigidly at a fixed speed.

usage (repo root): PYTHONPATH=.:visualization python noise_floor.py
"""
from pathlib import Path

import numpy as np
from ab_protocol_figures import deformation_stats, frame_shapes, sim_windows

from microbubble.initial_conditions import load_state

R = 1.78e-6
x0, _ = load_state(Path("inputs/pretrain_formed_states.h5"), "FORMED_N32_a1.0_s1")
for v_px in (43, 172):
    t = np.arange(0, 1.5 + 1e-9, 1 / 60)
    pos = np.array([x0 + np.array([v_px * 0.6e-6 * tt, 0, 0]) for tt in t])
    run = dict(times_s=t, positions_m=pos, segments=[dict(duration=1.5, amplitudes=[0, 0, 1, 0])])
    st = deformation_stats(sim_windows(run, frame_shapes(run, R), R))
    print(v_px, st["dln_aspect_per_px_first_1s"]["median"], st["dln_aspect_first_1s"]["median"])
===== scratch/adh2/figure_softness.py
"""Shape change per 0.1 s against cluster speed: A/B by speed band, simulations, rigid-body floor.

usage (repo root): PYTHONPATH=.:visualization python figure_softness.py MB_RESULTS ADHESION_MATCHED_SPEED.json FINAL_COMPARISON.json OUT.png
"""
import json
import sys
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
from ab_protocol_figures import real_cluster_windows, real_windows, speed_px_s

mb, adh, final, out = Path(sys.argv[1]), json.loads(Path(sys.argv[2]).read_text()), json.loads(Path(sys.argv[3]).read_text()), Path(sys.argv[4])
INK, INK2, MUTED, GRID, SURF = "#0b0b0b", "#52514e", "#8a8984", "#e6e5e1", "#fcfcfb"
REAL, NEW, OLD, FLOOR = "#eda100", "#1c5cab", "#eb6834", "#8a8984"
plt.rcParams.update({"font.family": "DejaVu Sans", "font.size": 9, "axes.edgecolor": MUTED,
                     "axes.labelcolor": INK2, "xtick.color": INK2, "ytick.color": INK2})
real = {e: real_windows(mb / d) for e, d in (("A", "A_offline"), ("B", "B_offline"))}
cl = real_cluster_windows(real)
on = [w for w in cl if w["action"] != "none" and w["t_since_s"] < 1.0 and w["dln_aspect"] is not None]
off = [w for w in cl if w["action"] == "none" and w["dln_aspect"] is not None]
fig, ax = plt.subplots(figsize=(7.4, 4.6))
fig.patch.set_facecolor(SURF)
ax.set_facecolor(SURF)
bands = [(0, 20), (20, 60), (60, 120), (120, 250), (250, 600)]
xs, ys = [], []
for lo, hi in bands:
    sel = [w for w in on if lo <= speed_px_s(w) < hi]
    if len(sel) >= 5:
        xs.append(np.median([speed_px_s(w) for w in sel]))
        ys.append(np.median([w["dln_aspect"] for w in sel]))
ax.plot(xs, ys, "-o", color=REAL, lw=2, ms=5, label="A/B clusters, PZT on (median per speed band)")
ax.plot([np.median([speed_px_s(w) for w in off])], [np.median([w["dln_aspect"] for w in off])], "D",
        color=REAL, mfc="white", ms=6, mew=1.6, label="A/B clusters, PZT off (tracking noise)")
g = final["groups"]
ax.plot(g["old protocol collective"]["speed_first_0p5s"], g["old protocol collective"]["dln_first_1s"], "s",
        color=OLD, ms=7, label="old: collective, grid start, no adhesion")
ax.plot(g["old protocol pairwise"]["speed_first_0p5s"], g["old protocol pairwise"]["dln_first_1s"], "s",
        color=MUTED, ms=7, label="old: pairwise")
ax.plot(g["new protocol N32"]["speed_first_0p5s"], g["new protocol N32"]["dln_first_1s"], "o", color=NEW,
        mfc="white", ms=7, mew=1.6, label="formed N32 + adhesion, 3000 Pa-eq")
first = True
for k, v in adh.items():
    if k.startswith("N32_k") and v["sim"]["aspect"] < 1.2:
        ax.plot(v["sim"]["speed"], v["sim"]["dln"], "o", color=NEW, ms=7, label="formed N32, 6000 Pa-eq, adhesion x0.1-1" if first else None)
        ax.plot(v["rigid_floor"]["speed"], v["rigid_floor"]["dln"], "x", color=FLOOR, ms=7, mew=1.6,
                label="same path, rigid cluster (pixel floor)" if first else None)
        first = False
v = adh["N128_k1"]
ax.plot(v["sim"]["speed"], v["sim"]["dln"], "^", color=NEW, ms=7, label="formed N128, 6000 Pa-eq")
ax.plot(v["rigid_floor"]["speed"], v["rigid_floor"]["dln"], "x", color=FLOOR, ms=7, mew=1.6)
ax.set_xscale("log")
ax.set_yscale("log")
ax.set_xlim(1, 600)
ax.set_ylim(1e-3, 0.3)
ax.set_xlabel("cluster speed (px/s)")
ax.set_ylabel("median |d ln aspect| per 0.1 s")
ax.set_title("Shape change against speed: formed clusters are now nearly rigid", loc="left", fontsize=9.5, color=INK)
for s in ("top", "right"):
    ax.spines[s].set_visible(False)
ax.grid(True, which="major", color=GRID, lw=0.5)
ax.legend(fontsize=6.8, frameon=False, loc="upper right")
fig.text(0.01, 0.005, "0.6 um/px rasterisation of visualization/ab_protocol_figures.py; X- held. Exploratory.",
         fontsize=6.8, color=INK2)
fig.tight_layout(rect=(0, 0.03, 1, 1))
fig.savefig(out, dpi=200, facecolor=SURF)
print(out)
===== scratch/speed/run_variant.py
"""Run a config with optional stepping overrides (scratch validation of the step control).

usage (repo root, PYTHONPATH=.): python run_variant.py CONFIG OUT [max_step_s=0.004,rejection_cap_factor=none]
"""
import json
import sys
from pathlib import Path

import workflows.f10ag as F

over = {}
for item in (sys.argv[3].split(",") if len(sys.argv) > 3 and sys.argv[3] not in ("", "{}") else []):
    key, value = item.split("=")
    over[key] = None if value == "none" else float(value)
original = F.load_pretrain_policy


def patched(root):
    p = original(root)
    p["stepping"] = dict(p["stepping"], **over)
    return p


F.load_pretrain_policy = patched
r = F.run(Path("."), Path(sys.argv[1]), Path(sys.argv[2]))
print(r["status"], r["completed_s"], round(r["wall_s"]), r["accepted_steps"], r["gate_rejected_steps"])
