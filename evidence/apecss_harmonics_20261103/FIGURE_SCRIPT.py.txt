"""Settled APECSS response vs pressure for several initial shell tensions (example grid run).

usage: response_grid_fig.py RESPONSE_GRID.h5 OUT.png
"""
import json
import sys
import numpy as np
import h5py
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

INK, INK2, MUTED, GRID, SURF = "#0b0b0b", "#52514e", "#8a8984", "#e6e5e1", "#fcfcfb"
COLORS = ["#2a78d6", "#eb6834", "#1baf7a", "#eda100", "#e87ba4", "#008300"]
plt.rcParams.update({"font.family": "DejaVu Sans", "font.size": 9, "axes.edgecolor": MUTED,
                     "axes.labelcolor": INK2, "xtick.color": INK2, "ytick.color": INK2})

with h5py.File(sys.argv[1]) as h:
    order = json.loads(h.attrs["axis_order"])
    ax = {k: h["axes/" + k][:] for k in order}
    H = h["harmonics_m3"][:]
    cycles = h["cycles"][:]
    status = h["status"][:]
assert order == ["sigma_initial_N_m", "frequency_Hz", "pressure_Pa"]
assert np.all(status == b"CONVERGED")
fi = int(np.flatnonzero(ax["frequency_Hz"] == 2.5e6)[0])
P = ax["pressure_Pa"] / 1000


def clean(a):
    a.set_facecolor(SURF)
    for s in ("top", "right"):
        a.spines[s].set_visible(False)
    a.grid(True, color=GRID, lw=0.6)
    a.set_axisbelow(True)


fig, axs = plt.subplots(1, 3, figsize=(14.5, 4.9), gridspec_kw=dict(wspace=0.32))
fig.patch.set_facecolor(SURF)
for a in axs:
    clean(a)
    a.set_xscale("log")
    a.set_xlabel("pressure amplitude at the bubble (kPa)")
    a.set_xlim(0.09, 75)
for si, s0 in enumerate(ax["sigma_initial_N_m"]):
    v1 = H[si, fi, :, 0]
    gain = np.abs(v1) / ax["pressure_Pa"]
    rel = gain / gain[0]
    phase = np.degrees(np.angle(v1))
    lab = f"σ₀ = {s0:g} N/m"
    axs[0].plot(P, rel, "o-", color=COLORS[si], lw=1.6, ms=3, label=lab)
    axs[1].plot(P, phase, "o-", color=COLORS[si], lw=1.6, ms=3, label=lab)
    axs[2].plot(P, cycles[si, fi], "o-", color=COLORS[si], lw=1.4, ms=3, label=lab)
axs[0].axhline(1, color=MUTED, lw=0.8)
axs[0].axhspan(0.95, 1.05, color=GRID, alpha=0.7, lw=0)
axs[0].text(0.1, 1.07, "±5 % band", color=INK2, fontsize=8)
axs[0].set_ylabel("pulsation per kPa, relative to its 0.1 kPa value")
axs[0].set_title("① How strongly it pulsates per unit pressure", loc="left", color=INK, fontsize=10)
axs[0].legend(frameon=False, fontsize=8, labelcolor=INK2, loc="upper left", bbox_to_anchor=(0.0, 0.93))
axs[1].axhline(90, color=INK, lw=0.9, ls="--")
axs[1].text(0.1, 93, "above 90°: pushed towards high pressure", color=INK2, fontsize=8)
axs[1].text(0.1, 80, "below 90°: pushed towards low pressure", color=INK2, fontsize=8)
axs[1].set_ylim(0, 180)
axs[1].set_ylabel("lag of the pulsation behind the pressure (deg)")
axs[1].set_title("② When it pulsates relative to the pressure", loc="left", color=INK, fontsize=10)
axs[2].axhline(48, color=MUTED, lw=1.2)
axs[2].text(0.1, 49.5, "fixed length of the frozen tables (48)", color=INK2, fontsize=8)
axs[2].set_ylim(0, 56)
axs[2].set_ylabel("cycles computed before stopping")
axs[2].set_title("③ Cycles needed (tolerance 1e-6)", loc="left", color=INK, fontsize=10)
fig.suptitle("Settled APECSS response of the 1.78 µm shelled bubble at 2.5 MHz for six initial shell tensions "
             "(other shell values as in the frozen tables, uncalibrated)", x=0.01, y=1.03, ha="left", color=INK, fontsize=11)
fig.savefig(sys.argv[2], dpi=150, facecolor=SURF, bbox_inches="tight")
