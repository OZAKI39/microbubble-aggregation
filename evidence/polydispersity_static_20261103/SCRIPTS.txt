# ===== chi_grid.py =====
"""Linear APECSS response chi(R) = V1/P for a grid of radii (same shell parameters as the frozen case).

usage (from the repository): PYTHONPATH=. python chi_grid.py APECSS_HARMONICS_EXE OUT.json
2.5 MHz, 1000 Pa (linear regime), ODE tolerance 1e-10, settled to 1e-8. Only the radius changes:
elasticity, shell viscosity, initial surface tension and all liquid/gas values stay those of the
1.78 um case (their radius dependence is unknown).
"""

import json
import sys
import tempfile
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

import numpy as np

from microbubble.bubble_response import apecss_harmonics, apecss_options

exe, out = sys.argv[1], Path(sys.argv[2])
case = json.loads(Path("inputs/response/2400000/PHYSICS_GATE.json").read_text())["response"]["case"]
radii = np.geomspace(0.5e-6, 6.0e-6, 161)
work = Path(tempfile.mkdtemp(prefix="chi_grid_", dir=str(out.parent)))


def one(k):
    opt = work / f"r{k:03d}.apecss"
    opt.write_text(apecss_options(dict(case, radius_m=float(radii[k])), 1e-10))
    r = apecss_harmonics(exe, opt, 2.5e6, 1000.0, tolerance=1e-8, max_cycles=2000)
    return r["status"], complex(r["harmonics_m3"][0]) / 1000.0, r["cycles"]


with ThreadPoolExecutor(2) as pool:
    rows = list(pool.map(one, range(len(radii))))
assert all(s == "CONVERGED" for s, _, _ in rows), [s for s, _, _ in rows]
chi = np.array([c for _, c, _ in rows])
out.write_text(json.dumps(dict(note=__doc__, radius_m=radii.tolist(), chi_re=chi.real.tolist(),
                               chi_im=chi.imag.tolist(), cycles=[c for _, _, c in rows]), indent=1) + "\n")
peak = radii[np.argmax(np.abs(chi) / radii ** 3)]
print("done", len(radii), "radius of largest |chi|/R^3:", peak)

# ===== poly_static.py =====
"""Static diagnostic: do bubbles of different sizes restore the cohesion of a compact cluster?

usage (from the repository): PYTHONPATH=. python poly_static.py CHI_GRID.json OUT.json
Clusters: monolayers (z = 0) packed around a centre with a 0.7 um surface gap (as AG32_local,
mean gap 0.71 um), radii monodisperse 1.78 um or lognormal with median 1.78 um and geometric
standard deviation s (clipped to 0.5-5.5 um, SonoVue diameters 1-11 um); also AG32_local itself.
Response: linear APECSS chi(R) (CHI_GRID.json, same shell parameters for every size).
Field: direct FNM at the bubbles, 1000 Pa-eq (linear regime): one X+ disc alone and the X pair.
Models: "isolated" (each bubble responds to the incident field only, like the pairwise model
without its two-bubble feedback) and "collective" (u = (I + chi K)^-1 chi P with the
propagation delay). Forces with the bare monopole formulas (no soft core) for every case:
F1_i = -1/2 Re[V_i conj(grad P_i)], pair force on i from j
-rho w^2 / (8 pi d^2) Re[V_i conj(V_j) e^{ikd}(1 - ikd)] r_hat(j->i).
Reported per cluster: median |u_i| / |chi_i P_i|; neighbour attraction (pairs with surface gap
< 1 um): median of the attractive component and fraction repulsive; spread of F1 about its
mean (rms); cohesion ratio = median neighbour attraction / rms F1 spread.
"""

import json
import sys
from pathlib import Path

import numpy as np

from microbubble.acoustics import direct_quadrature_order
from microbubble.initial_conditions import load_state
from workflows.f10ag import context

grid = json.load(open(sys.argv[1]))
out = Path(sys.argv[2])
RG = np.array(grid["radius_m"])
CHI = np.array(grid["chi_re"]) + 1j * np.array(grid["chi_im"])
R0, GAP, RHO = 1.78e-6, 0.7e-6, 1000.0
c = context(Path("."), 2.5e6, "coarse", "cpu")
K_WAVE = c.discs.k
W = 2 * np.pi * 2.5e6
ORDER = direct_quadrature_order(2.5e6)


def chi_of(r):
    lr = np.log(RG)
    return np.interp(np.log(r), lr, CHI.real) + 1j * np.interp(np.log(r), lr, CHI.imag)


def pack(radii, seed):
    """Monolayer of discs pulled to the centre; no two surfaces closer than GAP."""
    rng = np.random.default_rng(seed)
    n = len(radii)
    x = rng.normal(size=(n, 2)) * np.sqrt(n) * R0
    for it in range(4000):
        x -= 0.02 * x / (np.linalg.norm(x, axis=1, keepdims=True) + 1e-12) * R0 * (1 - it / 4000)
        for _ in range(4):
            d = x[:, None] - x[None]
            dist = np.linalg.norm(d, axis=-1) + np.eye(n)
            need = radii[:, None] + radii[None] + GAP
            over = np.clip(need - dist, 0, None) * (1 - np.eye(n))
            x += 0.5 * np.sum(over[:, :, None] * d / dist[:, :, None], axis=1)
    for _ in range(5000):  # final relaxation without the inward pull
        d = x[:, None] - x[None]
        dist = np.linalg.norm(d, axis=-1) + np.eye(n)
        need = radii[:, None] + radii[None] + GAP * 1.001
        over = np.clip(need - dist, 0, None) * (1 - np.eye(n))
        if not over.any():
            break
        x += 0.5 * np.sum(over[:, :, None] * d / dist[:, :, None], axis=1)
    d = np.linalg.norm(x[:, None] - x[None], axis=-1) + np.eye(n) * 1e3
    assert np.all(d - radii[:, None] - radii[None] >= GAP * 0.999)
    return np.column_stack([x, np.zeros(n)])


def analyse(x, radii, amps, centre):
    x = x - x.mean(0) + centre
    h, g = c.discs.evaluate(x, ORDER)
    usrc = c.basis.vector(amps, (0, 0, 0, 0))
    P = usrc @ h
    G = np.einsum("i,inj->nj", usrc, g)
    chi = chi_of(radii)
    n = len(x)
    D = np.linalg.norm(x[:, None] - x[None], axis=-1)
    off = ~np.eye(n, dtype=bool)
    K = np.where(off, RHO * W * W * np.exp(-1j * K_WAVE * D) / (4 * np.pi * np.where(off, D, 1)), 0)
    res = {}
    for model in ("isolated", "collective"):
        V = chi * P if model == "isolated" else np.linalg.solve(np.eye(n) + chi[:, None] * K, chi * P)
        F1 = -0.5 * np.real(V[:, None] * np.conj(G))
        i, j = np.triu_indices(n, 1)
        d = D[i, j]
        gap = d - radii[i] - radii[j]
        kd = K_WAVE * d
        att_i = RHO * W * W / (8 * np.pi * d * d) * np.real(V[i] * np.conj(V[j]) * np.exp(1j * kd) * (1 - 1j * kd))
        att_j = RHO * W * W / (8 * np.pi * d * d) * np.real(V[j] * np.conj(V[i]) * np.exp(1j * kd) * (1 - 1j * kd))
        m = gap < 1.0e-6
        att = np.concatenate([att_i[m], att_j[m]])
        spread = float(np.sqrt(np.mean(np.sum((F1 - F1.mean(0)) ** 2, axis=1))))
        res[model] = dict(
            suppression_median=float(np.median(np.abs(V) / np.abs(chi * P))),
            neighbour_pairs=int(m.sum()),
            attraction_median_N=float(np.median(att)),
            repulsive_fraction=float(np.mean(att < 0)),
            primary_spread_rms_N=spread,
            net_primary_N=float(np.linalg.norm(F1.sum(0))),
            cohesion_ratio=float(np.median(att) / spread),
        )
    return res


drives = {"X+ disc alone": ((0, 0, 0, 1000), np.array([3e-4, 1.5e-4, 0])),
          "X pair": ((0, 0, 1000, 1000), np.array([2e-5, 0, 0]))}
rows = []
x_ag, _ = load_state(Path("inputs/initial_states.h5"), "AG32_local")
cases = [("AG32_local (stored)", x_ag, np.full(32, R0), None)]
for n in (32, 96):
    for s in (1.0, 1.15, 1.3, 1.5):
        for seed in range(4):
            rng = np.random.default_rng(100 + seed)
            radii = np.full(n, R0) if s == 1.0 else np.clip(R0 * s ** rng.normal(size=n), 0.5e-6, 5.5e-6)
            cases.append((f"N{n} sigma_g {s}", pack(radii, seed), radii, dict(n=n, sigma_g=s, seed=seed)))
for name, x, radii, meta in cases:
    for dname, (amps, centre) in drives.items():
        r = analyse(x, radii, amps, centre)
        rows.append(dict(case=name, meta=meta, drive=dname, radius_min_um=float(radii.min() * 1e6),
                         radius_max_um=float(radii.max() * 1e6),
                         fraction_above_resonance_size=float(np.mean(radii > 2.05e-6)), **r))
        print(f"{name:22s} {dname:14s} | iso: att {r['isolated']['attraction_median_N']:+.2e} rep {r['isolated']['repulsive_fraction']:.2f} "
              f"ratio {r['isolated']['cohesion_ratio']:9.1f} | coll: supp {r['collective']['suppression_median']:.3f} "
              f"att {r['collective']['attraction_median_N']:+.2e} rep {r['collective']['repulsive_fraction']:.2f} ratio {r['collective']['cohesion_ratio']:8.2f}",
              flush=True)
out.write_text(json.dumps(dict(note=__doc__, rows=rows), indent=1) + "\n")

# ===== figure_poly.py =====
"""Figure: cohesion ratio vs size spread (static diagnostic)."""
import json
import sys
from collections import defaultdict

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np

INK, INK2, MUTED, GRID, SURF = "#0b0b0b", "#52514e", "#8a8984", "#e6e5e1", "#fcfcfb"
BLUE, ORANGE = "#2a78d6", "#eb6834"
plt.rcParams.update({"font.family": "DejaVu Sans", "font.size": 9, "axes.edgecolor": MUTED,
                     "axes.labelcolor": INK2, "xtick.color": INK2, "ytick.color": INK2})
rows = [r for f in sys.argv[1:-1] for r in json.load(open(f))["rows"] if r["meta"]]
g = defaultdict(list)
for r in rows:
    m = r["meta"]
    median = 1.25 if "median 1.25" in r["case"] else 1.78
    g[(median, m["n"], r["drive"], m["sigma_g"])].append(r)
fig, axes = plt.subplots(1, 2, figsize=(10.4, 4.2), sharey=True)
fig.patch.set_facecolor(SURF)
for ax, drive in zip(axes, ("X+ disc alone", "X pair")):
    ax.set_facecolor(SURF)
    for s_ in ("top", "right"):
        ax.spines[s_].set_visible(False)
    for (median, n, style) in ((1.78, 32, "-"), (1.78, 96, ":"), (1.25, 32, "--")):
        sig = sorted({k[3] for k in g if k[0] == median and k[1] == n and k[2] == drive})
        for model, color in (("isolated", ORANGE), ("collective", BLUE)):
            y = [max(np.median([r[model]["cohesion_ratio"] for r in g[(median, n, drive, s)]]), 0.01) for s in sig]
            ax.semilogy(sig, y, style, color=color, marker="o", ms=4, lw=1.7,
                        label=f"{model}, N {n}, median R {median} um")
    ax.axhline(1, color=MUTED, lw=0.8)
    ax.text(1.005, 0.3, "ratio 1: attraction = primary-force spread", color=INK2, fontsize=7.3)
    ax.set_xlabel("geometric standard deviation of bubble radius (1 = all equal)")
    ax.grid(True, which="both", color=GRID, lw=0.5)
    ax.set_axisbelow(True)
    ax.set_title(f"{'ab'[drive == 'X pair']}  {drive}, 1000 Pa-eq", loc="left", color=INK, fontsize=9.5)
axes[0].set_ylabel("neighbour attraction / primary-force spread")
h, l = axes[0].get_legend_handles_labels()
fig.legend(h, l, frameon=False, fontsize=7.6, labelcolor=INK2, loc="lower center", ncol=3, bbox_to_anchor=(0.5, -0.01), handlelength=3.2)
fig.tight_layout(rect=(0, 0.1, 1, 1))
fig.savefig(sys.argv[-1], dpi=200, facecolor=SURF)
