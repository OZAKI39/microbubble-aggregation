"""Carried APECSS response: 8 threads by default and the library compiled once (build/apecss/).

usage (repository root, PYTHONPATH=.:visualization, APECSS_DIR set):
  python apecss_speed_check.py short OUT_DIR     shortened runs with the defaults (cache warm)
  python apecss_speed_check.py runs OUT_DIR      whole runs in results/pretrain_apecss_response_threads8
  python apecss_speed_check.py figure OUT_DIR
Earlier numbers (1 thread, compiled every run) come from THREADS.json and RUN_COMPARISON.json.
Evidence script only; not imported by project code.
"""

import json
import sys
import tempfile
import time
from pathlib import Path

import h5py
import numpy as np

ROOT = Path.cwd()
RUNS = ("protocol_formed32", "protocol_formed128", "random_offset_B_formed32")
RUN_LABEL = {
    "protocol_formed32": "Protocol, 32 bubbles, 4.5 s",
    "protocol_formed128": "Protocol, 128 bubbles, 4.5 s",
    "random_offset_B_formed32": "Random actions, 32 bubbles, 2 s",
}


def short(out_dir):
    from workflows.f10ag import run

    rows = []
    with tempfile.TemporaryDirectory() as tmp:
        for name, duration_s in (("protocol_formed32", 0.5), ("protocol_formed128", 0.3)):
            for variant, extra in (("table", dict(response="table")), ("apecss_default", {})):
                cfg = json.loads((Path(out_dir) / f"configs/{name}_apecss.json").read_text())
                cfg["duration_s"] = duration_s
                cfg["control"]["segments"] = [dict(cfg["control"]["segments"][0], duration=duration_s)]
                path = Path(tmp) / f"{name}_{variant}.json"
                path.write_text(json.dumps(dict(cfg, **extra)))
                a = time.perf_counter()
                r = run(ROOT, path, Path(tmp) / f"{name}_{variant}")
                wall = time.perf_counter() - a
                ident = json.loads((Path(tmp) / f"{name}_{variant}" / "EXECUTION.json").read_text())["identity"]
                rec = ident.get("apecss_response")
                rows.append(dict(run=name, variant=variant, duration_s=duration_s, wall_s=wall,
                                 apecss_s=rec["stats"]["seconds_in_apecss"] if rec else 0.0,
                                 threads=rec["stats"]["settings"]["threads"] if rec else None,
                                 library_cached=rec["build"]["cached"] if rec else None,
                                 steps=r["constrained_solves"]))
                print(rows[-1])
    (Path(out_dir) / "SHORT_DEFAULTS.json").write_text(json.dumps(dict(
        rows=rows, note="same shortened runs as THREADS.json, with the defaults since the change "
                        "(8 APECSS threads, library from build/apecss/); one run at a time"), indent=1))


def runs(out_dir):
    new_base = ROOT / "results/pretrain_apecss_response_threads8"
    old_base = ROOT / "results/pretrain_apecss_response"
    result = {}
    for name in RUNS:
        row = {}
        x = {}
        for tag, base in (("old", old_base), ("new", new_base)):
            for response in ("table", "apecss"):
                d = base / f"{name}_{response}"
                s = json.loads((d / "SUMMARY.json").read_text())
                row[f"{tag}_{response}_wall_s"] = s["wall_s"]
                row[f"{tag}_{response}_status"] = s["status"]
                with h5py.File(d / "trajectory.h5") as h:
                    x[(tag, response)] = h["positions_m"][:]
        e = json.loads((new_base / f"{name}_apecss" / "EXECUTION.json").read_text())["identity"]["apecss_response"]
        row["new_apecss_threads"] = e["stats"]["settings"]["threads"]
        row["new_apecss_library_cached"] = e["build"]["cached"]
        row["new_apecss_seconds_in_apecss"] = e["stats"]["seconds_in_apecss"]
        row["apecss_new_identical_to_old"] = bool(np.array_equal(x[("new", "apecss")], x[("old", "apecss")]))
        row["table_new_identical_to_old"] = bool(np.array_equal(x[("new", "table")], x[("old", "table")]))
        result[name] = row
        print(name, row)
    (Path(out_dir) / "RUN_COMPARISON_THREADS8.json").write_text(json.dumps(dict(
        runs=result, note="old: 1 APECSS thread, library compiled at the start of every run; new: 8 threads, "
                          "library from build/apecss/; table and apecss twin runs side by side in both"), indent=1))


def figure(out_dir):
    import matplotlib

    matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    from pretrain_stability_figures import INK, setup_style

    out_dir = Path(out_dir)
    runs_ = json.loads((out_dir / "RUN_COMPARISON_THREADS8.json").read_text())["runs"]
    old_short = json.loads((out_dir / "THREADS.json").read_text())["rows"]
    new_short = json.loads((out_dir / "SHORT_DEFAULTS.json").read_text())["rows"]
    build_s = json.loads((out_dir / "SPEED_NOTES.json").read_text())["library_build_s"]
    orange, green, amber = "#eb6834", "#1baf7a", "#eda100"
    footer = ("Exploratory · pretraining tier · response \"apecss\" (carried APECSS bubbles), field \"direct_gpu\" · "
              "results identical for any thread count")
    with tempfile.TemporaryDirectory() as fonts:
        setup_style(Path(fonts))
        fig, axes = plt.subplots(1, 2, figsize=(13.5, 4.8))
        ax = axes[0]
        y = np.arange(len(RUNS))
        # Ratios inside one session each: the machine ran about 30% slower after a restart.
        bars = (("old", "Before: 1 thread, compiled every run", orange),
                ("new", "Now: 8 threads, compiled once", green))
        for k, (tag, label, color) in enumerate(bars):
            v = [runs_[n][f"{tag}_apecss_wall_s"] / runs_[n][f"{tag}_table_wall_s"] for n in RUNS]
            ax.barh(y + (k - 0.5) * 0.36, v, height=0.34, color=color, label=label)
            for yy, vv, n in zip(y, v, RUNS):
                a, t = runs_[n][f"{tag}_apecss_wall_s"], runs_[n][f"{tag}_table_wall_s"]
                ax.text(vv, yy + (k - 0.5) * 0.36, f" {vv:.2f}×  ({a:.0f} s vs {t:.0f} s)", va="center",
                        fontsize=8.5, color=INK["secondary"])
        ax.axvline(1, color=INK["muted"], linestyle="--", linewidth=1)
        ax.set_yticks(y, [RUN_LABEL[n].replace(", ", "\n", 1) for n in RUNS])
        ax.invert_yaxis()
        ax.set_xlim(0, 2.9)
        ax.set_xlabel("Run time with APECSS ÷ run time with the table (same session)")
        ax.set_title("a  Whole runs (table and APECSS twin side by side)")
        ax.legend(frameon=False, fontsize=8.5, loc="upper center", bbox_to_anchor=(0.5, -0.13), ncols=2)
        ax = axes[1]
        labels, parts = [], []
        for name, n_label in (("protocol_formed32", "32 bubbles, 0.5 s"), ("protocol_formed128", "128 bubbles, 0.3 s")):
            old = next(r for r in old_short if r["run"] == name and r["variant"] == "apecss_1")
            new = next(r for r in new_short if r["run"] == name and r["variant"] == "apecss_default")
            table = next(r for r in new_short if r["run"] == name and r["variant"] == "table")
            for tag, r, compile_s in (("APECSS before", old, build_s), ("APECSS now", new, 0.0), ("table", table, 0.0)):
                labels.append(f"{n_label}\n{tag}")
                parts.append((compile_s, r["apecss_s"], r["wall_s"] - compile_s - r["apecss_s"]))
        y = np.arange(len(labels))
        left = np.zeros(len(labels))
        for k, (label, color) in enumerate((("Compile APECSS library", amber), ("APECSS cycles", orange),
                                            ("Everything else", INK["muted"]))):
            v = np.array([p[k] for p in parts])
            ax.barh(y, v, left=left, color=color, height=0.6, label=label, edgecolor="white", linewidth=1)
            left += v
        for yy, total in zip(y, left):
            ax.text(total, yy, f" {total:.1f} s", va="center", fontsize=8.5, color=INK["secondary"])
        ax.set_yticks(y, labels, fontsize=8.5)
        ax.invert_yaxis()
        ax.set_xlim(0, left.max() * 1.3)
        ax.set_xlabel("Wall time of a shortened run (s, one run at a time)")
        ax.set_title("b  Where the time goes in a short run")
        ax.legend(frameon=False, fontsize=8.5, loc="lower right")
        fig.suptitle("Carried APECSS: 8 threads and a cached library", x=0.01, ha="left",
                     fontsize=12.5, fontweight="bold")
        fig.text(0.5, 0.008, footer, ha="center", va="bottom", fontsize=8, color=INK["muted"])
        fig.tight_layout(rect=(0, 0.05, 1, 0.95))
        fig.savefig(out_dir / "fig3_threads_and_cache.png", dpi=160)
        plt.close(fig)


if __name__ == "__main__":
    {"short": short, "runs": runs, "figure": figure}[sys.argv[1]](sys.argv[2])
